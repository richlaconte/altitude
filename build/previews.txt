function createFilesService(fixture = 'success-100', wait = (ms) => new Promise(resolve => setTimeout(resolve, ms))) {
  const records = new Map(FILES_SEED.map(record => [record.id, { ...record }]));
  const accepted = [], requests = [], receipts = new Map(), flights = new Map();
  let failedOnce = false;
  const read = id => records.has(id) ? { ...records.get(id) } : null;
  const error = (code, message) => Object.assign(new Error(message), { code });
  const change = (id, patch) => { const old = records.get(id); if (old) records.set(id, { ...old, ...patch, id, revision: old.revision + 1 }); };
  const apply = request => {
    const old = records.get(request.id);
    if (!old || old.deleted) throw error('missing', 'This file was removed. It cannot be restored.');
    if (old.revision !== request.expectedRevision) throw error('conflict', 'This file changed. Review its current version before trying again.');
    if (request.kind === 'delete' && !old.archived) throw error('conflict', 'This file is no longer in Archived. Review its current location.');
    const record = { ...old, revision: old.revision + 1 };
    if (request.kind === 'rename') record.name = request.name.trim();
    else if (request.kind === 'delete') record.deleted = true;
    else record.archived = request.kind === 'archive';
    records.set(record.id, record);
    accepted.push({ ...request, revision: record.revision });
    receipts.set(request.operationId, { ...record });
    return { ...record };
  };
  const mutate = request => {
    // Same logical request must not be submitted twice, even if the response was lost.
    if (receipts.has(request.operationId)) return Promise.resolve({ ...receipts.get(request.operationId) });
    if (flights.has(request.operationId)) return flights.get(request.operationId);
    requests.push({ ...request });

      if (mode === 'interrupted-after-save') {
        await wait(100); apply(request); await wait(1100);
        throw error('uncertain', 'Connection interrupted after save.');
      }
      const reject = mode === 'fail-once' && !failedOnce;

      await wait(mode === 'success-1200' || reject || mode === 'conflict' ? 1200 : 100);

    return flight;
  };

              <A.InlineEdit label="File name" editLabel="Rename file" value={file.name} onValueChange={publish('file-brand')} onSave={save('file-brand')} validate={validate('file-brand')} required maxLength={100} />
            </A.TableCell><A.TableCell>2.4 MB</A.TableCell></A.TableRow>}
            <A.TableRow><A.TableCell>{record('file-brand-draft').name}<p className="inline-demo-meta">Draft · Updated yesterday</p></A.TableCell><A.TableCell>1.8 MB</A.TableCell></A.TableRow>
            <A.TableRow><A.TableCell>{record('file-launch').name}</A.TableCell><A.TableCell>8 KB</A.TableCell></A.TableRow>
          </A.TableBody>
        </A.Table>
        <A.Button variant="ghost" onClick={() => setOutsideCount(count => count + 1)}>Show file details</A.Button>
        {outsideCount > 0 && <p className="inline-demo-meta">Three fictional files · Details opened {outsideCount} time{outsideCount === 1 ? '' : 's'}.</p>}
      </section>
    </div>
    <details className="inline-demo-harness">
      <summary>Comparison and fixture controls</summary>
      <p>These controls belong to the documentation harness. The dialog comparison uses the same filename, validation and save adapter. Reset before each trial.</p>
      <div className="inline-demo-controls">
        <A.Button onClick={externalUpdate}>Apply external filename update</A.Button>
        <A.Button onClick={() => { if (!visible) reconcile('file-brand'); setVisible(visible => !visible); }}>{visible ? 'Remove filename editor' : 'Show filename editor'}</A.Button>
        <A.Button onClick={() => setBaseline(true)}>Open dialog rename baseline</A.Button>
      </div>
      <output className="inline-demo-log" data-inline-log="" aria-label="Fixture operation log">{JSON.stringify(log, null, 2)}</output>
    </details>
    <InlineDialogBaseline open={baseline} onOpenChange={setBaseline} value={file.name} validate={validate('file-brand')} onSave={save('file-brand')} onValueChange={publish('file-brand')} />
  </>;
}

function InlineDialogBaseline({ open, onOpenChange, value, validate, onSave, onValueChange }) {
  const [draft, setDraft] = React.useState(value), [error, setError] = React.useState(''), [busy, setBusy] = React.useState('');
  const baseline = React.useRef(value), generation = React.useRef(0), guard = React.useRef(false), latest = React.useRef(value);
  latest.current = value;
  React.useEffect(() => {
    generation.current++; guard.current = false; setBusy(''); setError('');
    if (open) { baseline.current = value; setDraft(value); }
    return () => { generation.current++; };
  }, [open]);
  const submit = async () => {
    if (guard.current) return;
    if (latest.current !== baseline.current) { setError('The saved value changed. Close and reopen to review it.'); return; }
    if (draft === baseline.current) { onOpenChange(false); return; }
    guard.current = true; const token = generation.current; setError(''); setBusy('Checking value');
    const error = draft.length > 100 ? 'Use 100 characters or fewer.' : await validate(draft);
    if (token !== generation.current) return;
    if (error) { guard.current = false; setBusy(''); setError(error); return; }
    setBusy('Saving changes');
    const result = await onSave(draft, { previousValue: baseline.current });
    if (token !== generation.current) return;
    guard.current = false; setBusy('');
    if (latest.current !== baseline.current) { setError('The value changed while saving. Your save may have completed. Check the current record.'); return; }
    if (!result.ok) { setError(result.error); return; }
    onValueChange(result.value); onOpenChange(false);
  };
  return <A.Dialog open={open} onOpenChange={onOpenChange} title="Rename file" description="Change the name of this file." dismissible={busy !== 'Saving changes'} footer={<>
    <A.Button variant="ghost" disabled={busy === 'Saving changes'} onClick={() => onOpenChange(false)}>Cancel</A.Button>
    <A.Button variant="primary" loading={busy === 'Saving changes'} disabled={busy === 'Checking value'} onClick={submit}>Save changes</A.Button>
  </>}>
    <A.Field label="File name" error={error} required><A.Input value={draft} maxLength={100} readOnly={!!busy} onChange={event => { setDraft(event.target.value); setError(''); }} onKeyDown={event => {
      if (event.nativeEvent.isComposing || event.nativeEvent.keyCode === 229) return;
      if (event.key === 'Enter') { event.preventDefault(); submit(); }
    }} /></A.Field>
    <p role="status">{busy}</p>
  </A.Dialog>;
}

function App() {
  const [fixture, setFixture] = React.useState('success-100'), [slow, setSlow] = React.useState(false);
  const [mounted, setMounted] = React.useState({ fixture: 'success-100', slow: false, epoch: 0 });
  return <div className="inline-demo">
    <InlineExample key={mounted.epoch} fixture={mounted.fixture} slowValidation={mounted.slow} epoch={mounted.epoch} />
    <details className="inline-demo-harness">
      <summary>Choose the next fixture</summary>
      <p>Select before mounting. Reset restores the same fictional records, revisions and request counters.</p>
      <div className="inline-demo-controls">
        <label>Rename fixture<select value={fixture} onChange={event => setFixture(event.target.value)}>{INLINE_FIXTURES.map(name => <option key={name}>{name}</option>)}</select></label>
        <label><input type="checkbox" checked={slow} onChange={event => setSlow(event.target.checked)} />Delay validation by 1,200 ms</label>
        <A.Button onClick={() => setMounted({ fixture, slow, epoch: mounted.epoch + 1 })}>Reset example</A.Button>
      </div>
    </details>
  </div>;
}
